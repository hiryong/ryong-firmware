cmake_minimum_required(VERSION 3.16)
project(ryong_firmware_host_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()

add_library(ryong_core STATIC src/ryong_core.c)
target_include_directories(ryong_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(MSVC)
  target_compile_options(ryong_core PRIVATE /W4)
else()
  target_compile_options(ryong_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(ryong_core_tests tests/test_core.c)
target_link_libraries(ryong_core_tests PRIVATE ryong_core)
add_test(NAME ryong_core_tests COMMAND ryong_core_tests)

add_library(ryong_host_policy STATIC src/ryong_host_policy.c)
target_include_directories(ryong_host_policy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(NOT MSVC)
  target_compile_options(ryong_host_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(ryong_host_policy_tests tests/test_host_policy.c)
target_link_libraries(ryong_host_policy_tests PRIVATE ryong_host_policy)
add_test(NAME ryong_host_policy_tests COMMAND ryong_host_policy_tests)


add_library(ryong_host_layout STATIC modules/ryong-host-layout/src/host_layout.c)
target_include_directories(ryong_host_layout PUBLIC modules/ryong-host-layout/include)
if(NOT MSVC)
  target_compile_options(ryong_host_layout PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(ryong_host_layout_tests tests/test_host_layout.c)
target_link_libraries(ryong_host_layout_tests PRIVATE ryong_host_layout)
add_test(NAME ryong_host_layout_tests COMMAND ryong_host_layout_tests)
